// sd_spi.c
#include "u_spi_flash.h"

// Global SD card structure
static sd_card_t sd_card = {0};

// Internal helper function prototypes
static uint8_t sd_send_command(uint8_t cmd, uint32_t arg);
static uint8_t sd_send_app_command(uint8_t cmd, uint32_t arg);
static int sd_read_data(uint8_t *buffer, uint16_t len);
static int sd_write_data(const uint8_t *buffer, uint8_t token);
static void sd_deselect(void);
static void sd_select(void);
static uint8_t sd_wait_ready(uint32_t timeout_ms);

// Internal helper implementations

static void sd_select(void)
{
	digitalWrite(SPI_CS_PIN, LOW);
	delay_ms(1);
}

static void sd_deselect(void)
{
	digitalWrite(SPI_CS_PIN, HIGH);
	spi_transfer(0xFF); // Extra clock cycles
}

static uint8_t sd_wait_ready(uint32_t timeout_ms)
{
    uint8_t res;
    
    // Fast polling - no delay
    for (uint32_t i = 0; i < (timeout_ms * 10000); i++) {
        res = spi_transfer(0xFF);
        if (res == 0xFF) return 1;
    }
    
    return 0;
}

static uint8_t sd_send_command(uint8_t cmd, uint32_t arg)
{
	uint8_t crc = 0xFF;
	uint8_t res;
	int retries = 10;

	// Special CRC for CMD0 and CMD8
	if (cmd == CMD0)
		crc = 0x95;
	if (cmd == CMD8)
		crc = 0x87;

	// Wait for card to be ready
	sd_wait_ready(300);

	// Send command
	spi_transfer(0x40 | cmd);
	spi_transfer((arg >> 24) & 0xFF);
	spi_transfer((arg >> 16) & 0xFF);
	spi_transfer((arg >> 8) & 0xFF);
	spi_transfer(arg & 0xFF);
	spi_transfer(crc);

	// Wait for response (not 0xFF)
	do
	{
		res = spi_transfer(0xFF);
		if (!(res & 0x80))
			break;
	} while (--retries);

	return res;
}

static uint8_t sd_send_app_command(uint8_t cmd, uint32_t arg)
{
	sd_send_command(CMD55, 0);
	return sd_send_command(cmd, arg);
}

static int sd_read_data(uint8_t *buffer, uint16_t len)
{
	uint16_t timeout = 5000;
	uint8_t token;

	// Wait for data token
	do
	{
		token = spi_transfer(0xFF);
		if (token == DATA_START_BLOCK)
			break;
		delay_ms(1);
	} while (--timeout);

	if (token != DATA_START_BLOCK)
	{
		uart_print("SD: Read timeout\r\n");
		return SD_ERROR_TIMEOUT;
	}

	// Read data
	for (uint16_t i = 0; i < len; i++)
	{
		buffer[i] = spi_transfer(0xFF);
	}

	// Read CRC (2 bytes, ignored)
	spi_transfer(0xFF);
	spi_transfer(0xFF);

	return SD_OK;
}

static int sd_write_data(const uint8_t *buffer, uint8_t token)
{
	uint8_t res;

	if (!sd_wait_ready(500))
	{
		uart_print("SD: Card not ready for write\r\n");
		return SD_ERROR_WRITE;
	}

	// Send token
	spi_transfer(token);

	if (token != STOP_TRAN_TOKEN)
	{
		// Send data block (512 bytes)
		for (uint16_t i = 0; i < 512; i++)
		{
			spi_transfer(buffer[i]);
		}

		// Send dummy CRC
		spi_transfer(0xFF);
		spi_transfer(0xFF);

		// Get data response
		res = spi_transfer(0xFF);
		if ((res & 0x1F) != DATA_ACCEPTED)
		{
			uart_print("SD: Write rejected: 0x");
			uart_print_hex8(res);
			uart_print("\r\n");
			return SD_ERROR_WRITE;
		}

		// Wait for write to complete
		if (!sd_wait_ready(500))
		{
			uart_print("SD: Write timeout\r\n");
			return SD_ERROR_TIMEOUT;
		}
	}

	return SD_OK;
}

// Public API implementations

int sd_init(void)
{
	uint8_t res;
	int retries;
	uint8_t ocr[4];

	uart_print("SD: Initializing...\r\n");

	// Initialize SPI
	spi_init();
	spi_set_clock(512);
	gpio_set_function(SPI_CS_PIN, GPIO_FUNC_OUTPUT);
	digitalWrite(SPI_CS_PIN, HIGH);

	// Send 80+ clock cycles with CS high
	for (int i = 0; i < 10; i++)
	{
		spi_transfer(0xFF);
	}
	delay_ms(10);

	// Select card
	sd_select();

	// CMD0: Reset card
	retries = 10;
	do
	{
		res = sd_send_command(CMD0, 0);
		if (res == R1_IDLE_STATE)
			break;
		delay_ms(100);
	} while (--retries);

	if (res != R1_IDLE_STATE)
	{
		uart_print("SD: CMD0 failed: 0x");
		uart_print_hex8(res);
		uart_print("\r\n");
		sd_deselect();
		return SD_ERROR_INIT;
	}

	uart_print("SD: Card in idle state\r\n");

	// CMD8: Check voltage range (SD V2.0+)
	res = sd_send_command(CMD8, 0x1AA);

	if (res == R1_IDLE_STATE)
	{
		// SD V2.0+
		uart_print("SD: SD V2.0+ detected\r\n");

		// Read R7 response (4 bytes)
		for (int i = 0; i < 4; i++)
		{
			ocr[i] = spi_transfer(0xFF);
		}

		// Check if voltage accepted
		if ((ocr[2] & 0x0F) != 0x01 || ocr[3] != 0xAA)
		{
			uart_print("SD: Voltage range not supported\r\n");
			sd_deselect();
			return SD_ERROR_INIT;
		}

		// ACMD41: Initialize card with HCS bit
		retries = 1000;
		do
		{
			res = sd_send_app_command(ACMD41, 0x40000000);
			if (res == R1_READY_STATE)
				break;
			delay_ms(10);
		} while (--retries);

		if (res != R1_READY_STATE)
		{
			uart_print("SD: ACMD41 failed\r\n");
			sd_deselect();
			return SD_ERROR_INIT;
		}

		// CMD58: Read OCR to check CCS bit
		res = sd_send_command(CMD58, 0);
		if (res != 0)
		{
			uart_print("SD: CMD58 failed\r\n");
			sd_deselect();
			return SD_ERROR_INIT;
		}

		for (int i = 0; i < 4; i++)
		{
			ocr[i] = spi_transfer(0xFF);
		}

		// Check CCS bit (bit 30)
		if (ocr[0] & 0x40)
		{
			sd_card.type = SD_CARD_TYPE_SDHC;
			uart_print("SD: SDHC/SDXC card detected\r\n");
		}
		else
		{
			sd_card.type = SD_CARD_TYPE_SD2;
			uart_print("SD: SD V2.0 card detected\r\n");
		}
	}
	else
	{
		// SD V1.0 or MMC
		uart_print("SD: SD V1.0 or MMC detected\r\n");
		sd_card.type = SD_CARD_TYPE_SD1;

		// ACMD41: Initialize SD V1.0
		retries = 1000;
		do
		{
			res = sd_send_app_command(ACMD41, 0);
			if (res == R1_READY_STATE)
				break;
			delay_ms(10);
		} while (--retries);

		if (res != R1_READY_STATE)
		{
			uart_print("SD: ACMD41 failed\r\n");
			sd_deselect();
			return SD_ERROR_INIT;
		}
	}

	// Set block size to 512 bytes (for non-SDHC cards)
	if (sd_card.type != SD_CARD_TYPE_SDHC)
	{
		res = sd_send_command(CMD16, 512);
		if (res != 0)
		{
			uart_print("SD: CMD16 failed\r\n");
			sd_deselect();
			return SD_ERROR_INIT;
		}
	}

	sd_card.block_size = 512;

	// Read CSD to get capacity
	res = sd_send_command(CMD9, 0);
	if (res != 0)
	{
		uart_print("SD: CMD9 failed\r\n");
		sd_deselect();
		return SD_ERROR_INIT;
	}

	uint8_t csd[16];
	if (sd_read_data(csd, 16) != SD_OK)
	{
		uart_print("SD: CSD read failed\r\n");
		sd_deselect();
		return SD_ERROR_INIT;
	}

	// Calculate capacity based on CSD structure
	if (sd_card.type == SD_CARD_TYPE_SDHC)
	{
		// CSD V2.0 (SDHC/SDXC)
		uint32_t c_size = ((uint32_t)(csd[7] & 0x3F) << 16) |
						  ((uint32_t)csd[8] << 8) | csd[9];
		sd_card.capacity = (c_size + 1) * 1024; // in blocks
	}
	else
	{
		// CSD V1.0
		uint32_t c_size = ((uint32_t)(csd[6] & 0x03) << 10) |
						  ((uint32_t)csd[7] << 2) |
						  ((csd[8] & 0xC0) >> 6);
		uint32_t c_size_mult = ((csd[9] & 0x03) << 1) | ((csd[10] & 0x80) >> 7);
		uint32_t read_bl_len = csd[5] & 0x0F;
		uint32_t mult = 1 << (c_size_mult + 2);
		uint32_t block_len = 1 << read_bl_len;
		sd_card.capacity = (c_size + 1) * mult * block_len / 512;
	}

	sd_card.size_bytes = (uint64_t)sd_card.capacity * 512;
	sd_card.initialized = 1;

	spi_set_clock(28);
	uart_print("SD: SPI speed increased\r\n");

	sd_deselect();

	uart_print("SD: Initialized successfully\r\n");
	uart_print("SD: Capacity: ");
	char tmpstrbuf[30];
	uart_print(ulltoa(sd_card.size_bytes, tmpstrbuf));
	uart_print(" bytes (");
	uart_print(ulltoa(sd_card.size_bytes / (1024 * 1024), tmpstrbuf));
	uart_print(" MB)\r\n");
	uart_print("SD: Blocks: ");
	uart_print(ulltoa(sd_card.capacity, tmpstrbuf));
	uart_print("\r\n");

	return SD_OK;
}

uint64_t sd_get_size(void)
{
	return sd_card.size_bytes;
}

uint32_t sd_get_block_count(void)
{
	return sd_card.capacity;
}

int sd_read_block(uint32_t block, uint8_t *buffer)
{
	uint8_t res;

	if (!sd_card.initialized)
	{
		return SD_ERROR_INIT;
	}

	// Convert block number to byte address for non-SDHC cards
	uint32_t addr = (sd_card.type == SD_CARD_TYPE_SDHC) ? block : block * 512;

	sd_select();

	res = sd_send_command(CMD17, addr);
	if (res != 0)
	{
		uart_print("SD: CMD17 failed: 0x");
		uart_print_hex8(res);
		uart_print("\r\n");
		sd_deselect();
		return SD_ERROR_READ;
	}

	int ret = sd_read_data(buffer, 512);

	sd_deselect();

	return ret;
}

int sd_write_block(uint32_t block, const uint8_t *buffer)
{
	uint8_t res;

	if (!sd_card.initialized)
	{
		return SD_ERROR_INIT;
	}

	uint32_t addr = (sd_card.type == SD_CARD_TYPE_SDHC) ? block : block * 512;

	sd_select();

	res = sd_send_command(CMD24, addr);
	if (res != 0)
	{
		uart_print("SD: CMD24 failed: 0x");
		uart_print_hex8(res);
		uart_print("\r\n");
		sd_deselect();
		return SD_ERROR_WRITE;
	}

	int ret = sd_write_data(buffer, DATA_START_BLOCK);

	sd_deselect();

	return ret;
}

int sd_read_multiple_blocks(uint32_t block, uint8_t *buffer, uint32_t count)
{
	uint8_t res;

	if (!sd_card.initialized)
	{
		return SD_ERROR_INIT;
	}

	uint32_t addr = (sd_card.type == SD_CARD_TYPE_SDHC) ? block : block * 512;

	sd_select();

	res = sd_send_command(CMD18, addr);
	if (res != 0)
	{
		uart_print("SD: CMD18 failed\r\n");
		sd_deselect();
		return SD_ERROR_READ;
	}

	for (uint32_t i = 0; i < count; i++)
	{
		if (sd_read_data(buffer + (i * 512), 512) != SD_OK)
		{
			sd_send_command(CMD12, 0); // Stop transmission
			sd_deselect();
			return SD_ERROR_READ;
		}
	}

	sd_send_command(CMD12, 0); // Stop transmission
	sd_deselect();

	return SD_OK;
}

int sd_write_multiple_blocks(uint32_t block, const uint8_t *buffer, uint32_t count)
{
	uint8_t res;

	if (!sd_card.initialized)
	{
		return SD_ERROR_INIT;
	}

	uint32_t addr = (sd_card.type == SD_CARD_TYPE_SDHC) ? block : block * 512;

	sd_select();

	res = sd_send_command(CMD25, addr);
	if (res != 0)
	{
		uart_print("SD: CMD25 failed\r\n");
		sd_deselect();
		return SD_ERROR_WRITE;
	}

	for (uint32_t i = 0; i < count; i++)
	{
		if (sd_write_data(buffer + (i * 512), DATA_START_BLOCK_MULTI) != SD_OK)
		{
			sd_write_data(NULL, STOP_TRAN_TOKEN);
			sd_deselect();
			return SD_ERROR_WRITE;
		}
	}

	sd_write_data(NULL, STOP_TRAN_TOKEN);
	sd_deselect();

	return SD_OK;
}

int sd_erase_blocks(uint32_t start_block, uint32_t end_block)
{
	uint8_t res;

	if (!sd_card.initialized)
	{
		return SD_ERROR_INIT;
	}

	uint32_t start_addr = (sd_card.type == SD_CARD_TYPE_SDHC) ? start_block : start_block * 512;
	uint32_t end_addr = (sd_card.type == SD_CARD_TYPE_SDHC) ? end_block : end_block * 512;

	sd_select();

	// Set start address
	res = sd_send_command(CMD32, start_addr);
	if (res != 0)
	{
		uart_print("SD: CMD32 failed\r\n");
		sd_deselect();
		return SD_ERROR_ERASE;
	}

	// Set end address
	res = sd_send_command(CMD33, end_addr);
	if (res != 0)
	{
		uart_print("SD: CMD33 failed\r\n");
		sd_deselect();
		return SD_ERROR_ERASE;
	}

	// Erase
	res = sd_send_command(CMD38, 0);
	if (res != 0)
	{
		uart_print("SD: CMD38 failed\r\n");
		sd_deselect();
		return SD_ERROR_ERASE;
	}

	// Wait for erase to complete (can take a while)
	if (!sd_wait_ready(5000))
	{
		uart_print("SD: Erase timeout\r\n");
		sd_deselect();
		return SD_ERROR_TIMEOUT;
	}

	sd_deselect();

	uart_print("SD: Erased blocks ");
	char tmpstrbuf[30];
	uart_print(ulltoa(start_block, tmpstrbuf));
	uart_print(" to ");
	uart_print(ulltoa(end_block, tmpstrbuf));
	uart_print("\r\n");

	return SD_OK;
}
